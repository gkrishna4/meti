name: Backup

on:
  workflow_dispatch:
    inputs:
      account:
        type: choice
        description: Please choose the account (environment) to create a backup for.
        required: true
        options:
        - dev1
        - dev2
        - dev3
        - int
        - prod
      tenants:
        description: Please provide the tenant ids (lowercase short form as JSON array) to create a backup for. (Example:'["pt"]').
        required: true
        default: '["pt"]'
      ticketnumber:
        description: Please enter the ticket number related to this backup
        type: string
        required: true

jobs:
  setEnvVars:
    runs-on: self-hosted
    outputs:
      AccountID: ${{ steps.set-env-id.outputs.AccountID }}
    environment: ${{ inputs.Environment }}
    steps:
      - name: checkout
        uses: actions/checkout@v2
      - name: set environment id
        id: set-env-id
        env:
          ICON_ENVIRONMENT: "${{ inputs.account }}"
        run: |
          account_id=$(jq -cr --arg ICON_ENV "${ICON_ENVIRONMENT}" '.[$ICON_ENV].account_id' ./.github/workflows/account_id_config.json)
          stage=$(jq -cr --arg ICON_ENV "${ICON_ENVIRONMENT}" '.[$ICON_ENV].stage' ./.github/workflows/account_id_config.json)
          echo "::set-output name=AccountID::"$account_id""

  backup:
    strategy:
      matrix:
        tenant: ${{ fromJSON(github.event.inputs.tenants) }}
      fail-fast: false
    concurrency: "${{ inputs.Environment }}-${{ matrix.tenant }}"
    needs: setEnvVars
    runs-on:
      - self-hosted
      - ${{ needs.setEnvVars.outputs.AccountID }}
    permissions:
      id-token: write
      contents: read
    steps:
    - name: checkout
      uses: actions/checkout@v2
    - name: configure aws credentials
      uses: aws-actions/configure-aws-credentials@v2
      with:
        role-to-assume: "arn:aws:iam::${{ needs.setEnvVars.outputs.AccountID }}:role/icon-gha-workflow-role"
        role-session-name: "backup-${{ inputs.Environment }}"
        role-duration-seconds: 7200
        aws-region: "eu-central-1"
    - name: create backup
      env:
        ICON_TICKETNUMBER: "${{ inputs.ticketnumber }}"
      run: |
        echo "Tenant: ${{ matrix.tenant }}"
        echo "Ticketnumber: ${ICON_TICKETNUMBER}"

        readonly job_name="backup-${{ matrix.tenant }}-$(openssl rand -hex 12)"
        readonly backup_job_id="$(aws batch submit-job --job-name "$job_name" --job-queue "db-job-queue" \
        --job-definition "db-backup-${{ matrix.tenant }}" \
        --container-overrides "{\"environment\": [{\"name\":\"ICON_BACKUP_TYPE\",\"value\":\"long-term/${ICON_TICKETNUMBER}\"}]}" | jq -r '.jobId')"

        echo "Job name: ${job_name}"
        echo "Job ID: ${backup_job_id}"

        jobs_status=''

        echo "Waiting for job: ${job_name}"

        while [[ ! "${jobs_status}" =~ (SUCCEEDED|FAILED|COMPLETED) ]]; do
            jobs_status="$(aws batch describe-jobs --jobs "${backup_job_id}" | jq -r '.jobs[0].status')"
            echo "Still waiting for job ${job_name}... Current status is ${jobs_status}."
            sleep 5
        done

        if [[ "${jobs_status}" == "FAILED" ]]; then
          echo "Backup failed. Aborting..."
          exit 1
        fi

            exit 1
          else
            echo "Waiting for $delta seconds before continuing..."

      uses: aws-actions/configure-aws-credentials@v2
      with:
        role-to-assume: arn:aws:iam::${{ needs.init.outputs.sourceAccountId }}:role/icon-gha-workflow-role
        role-session-name: "find-backups-${{ inputs.Environment }}"
        aws-region: eu-central-1
    - name: Identify backups
      id: identify_backups
      env: 
        ICON_TENANTS: "${{ inputs.tenants }}"
        ICON_BACKUP_TIMESTAMP: "${{ inputs.timestamp }}"
        ICON_SOURCE_BUCKETNAME: "${{ needs.init.outputs.sourceBucketName }}"
      run: |
        echo "Tenants: ${ICON_TENANTS}"
        echo "Source buckets: ${ICON_SOURCE_BUCKETNAME}"
        echo "Timestamp used for backup search: ${ICON_BACKUP_TIMESTAMP}"

        found_backups='{}'

        for tenant_id_short in "$(jq -r '.[]' <<< "${ICON_TENANTS}")"; do
          tenant_id="$(jq -cr --arg TENANT_ID "${tenant_id_short}" '.[$TENANT_ID].tenantid' ./.github/workflows/tenant_config.json)"

          found_keys="$(aws --query "Contents[?contains(Key, '${ICON_BACKUP_TIMESTAMP}')].Key" s3api list-objects-v2 \
            --prefix "db-backups/${tenant_id}/" --bucket "${ICON_SOURCE_BUCKETNAME}" | jq -c '. // []')"

          count_of_found_keys="$(jq '. | length' <<< "$found_keys")"
          backup_key=''

          if [[ ${count_of_found_keys} -eq 0 ]]; then
              echo "No backup found for tenant ${tenant_id_short}."
              echo "Exiting"
              exit 1
          elif [ ${count_of_found_keys} -gt 1 ]; then
              echo "Multiple backups found for tenant ${tenant_id_short}."
              echo "${found_keys}"
              echo "Exiting"
              exit 1
          else
              backup_key="$(jq -r '.[0]' <<< "$found_keys")"
              echo "Backup found and used for restore of tenant ${tenant_id_short}: ${backup_key}"
              found_backups="$(jq -c --arg TENANT_ID_SHORT "${tenant_id_short}" --arg BACKUP_KEY "${backup_key}" '. += {($TENANT_ID_SHORT): $BACKUP_KEY}' <<< $found_backups)"
          fi
        done
        echo "Found backups: ${found_backups}"
        echo "::set-output name=backup_s3_keys::"${found_backups}""

  approval:
    needs: [init, FindBackups]
    uses: ./.github/workflows/ApprovalReusable.yml
    with:
      target: ${{ inputs.targetEnvironment }}-restore
      restore_anonymize : ${{ inputs.anonymize }}

  Restore:
    if: |
      !cancelled() &&
      needs.FindBackups.result == 'success' &&
      needs.approval.result != 'failure' &&
      needs.approval.result != 'cancelled'
    strategy:
      matrix:
        tenant: ${{ fromJSON(github.event.inputs.tenants) }}
      fail-fast: false
    concurrency: "${{ inputs.Environment }}-${{ matrix.tenant }}"
    needs: ['approval','init','FindBackups']
    runs-on:
      - self-hosted
      - ${{ needs.init.outputs.targetEnvironmentId }}
    permissions:
      id-token: write
      contents: read
    env:
      ICON_TENANT: "${{ matrix.tenant }}"
    steps:
      - name: Checkout
        uses: actions/checkout@v2
      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v2
        with:
          role-to-assume: arn:aws:iam::${{ needs.init.outputs.targetEnvironmentId }}:role/icon-gha-workflow-role
          role-session-name: "restore-${{ inputs.Environment }}"
          role-duration-seconds: 28800
          aws-region: "eu-central-1"
      - name: Stop iCON services
        id: stop_services
        run: |
            nemesis_replicas="$(aws ecs describe-services --cluster "icon-cluster" --services "icon-nemesis-service-${ICON_TENANT,,}" --query 'services[0].desiredCount')"
            online_replicas="$(aws ecs describe-services --cluster "icon-cluster" --services "icon-online-service-${ICON_TENANT,,}" --query 'services[0].desiredCount')"
            echo "::set-output name=nemesis_replicas::"${nemesis_replicas}""
            echo "::set-output name=online_replicas::"${online_replicas}""

            echo "Stopping icon-nemesis-service for tenant: ${ICON_TENANT,,}"
            aws ecs update-service --cluster "icon-cluster" --service "icon-nemesis-service-${ICON_TENANT,,}" --desired-count "0" > /dev/null
            aws ecs wait services-stable --cluster "icon-cluster" --service "icon-nemesis-service-${ICON_TENANT,,}"
            echo "Stopping icon-online-service for tenant: ${ICON_TENANT,,}"
            aws ecs update-service --cluster "icon-cluster" --service "icon-online-service-${ICON_TENANT,,}" --desired-count "0" > /dev/null
            aws ecs wait services-stable --cluster "icon-cluster" --service "icon-online-service-${ICON_TENANT,,}"
      - name: Disable batch queue
        run: |
            echo "Disabling batches for tenant: ${ICON_TENANT,,}"
            bash ./icon-infrastructure/automation/actions-scripts/disable_batch_queue.sh -q "icon-queue-${ICON_TENANT,,}"
      - name: Restore
        env:
          ICON_SOURCE_ACCOUNT_ID: "${{ needs.init.outputs.sourceAccountId }}"
          ICON_TARGET_ACCOUNT_ID: "${{ needs.init.outputs.targetEnvironmentId }}"
          ICON_SOURCE_BUCKETNAME: "${{ needs.init.outputs.sourceBucketName }}"
          ICON_AFTER_RR: "${{ inputs.after_rr_procedure }}"
          ICON_ANONYMIZE: "${{ inputs.anonymize }}"
          ICON_VERIFY: "${{ inputs.verification }}"
          ICON_VERIFY_DATE: "${{ inputs.verification_date }}"
          ICON_BACKUPS_FOUND: "${{ needs.FindBackups.outputs.backupsFound }}"
        run: |
          set -eE
          backup_key="$(jq -cr --arg TENANT_ID_SHORT "${ICON_TENANT,,}" '.[$TENANT_ID_SHORT]' <<< "${ICON_BACKUPS_FOUND}")"
          if [[ -z "$backup_key" ]]; then
            echo "Backupfile could not be identified. Skipping restore..."
            exit 0
          fi

          echo "Running restore with backup: "${backup_key}""
          echo "After RR procedure: ${ICON_AFTER_RR}"
          echo "Anonymize: ${ICON_ANONYMIZE}"

          readonly job_name="restore-${ICON_TENANT,,}-$(openssl rand -hex 12)"

          assume_role_arg=''
          if [[ ! "${ICON_SOURCE_ACCOUNT_ID}" == "${ICON_TARGET_ACCOUNT_ID}" ]]; then
            assume_role_arg=",\"-a\",\"arn:aws:iam::${ICON_SOURCE_ACCOUNT_ID}:role/db-job-batch-cross-account-source-s3-role\""
          fi

          varify_arg=''
          if [[ "${ICON_VERIFY}" == "true" ]]; then
            varify_arg=",\"-q\""
          fi

          command_args="[\"-k\",\"${backup_key}\",\"-b\",\"${ICON_SOURCE_BUCKETNAME}\"${assume_role_arg}${varify_arg}]"
          environment_args="[{\"name\":\"ICON_AFTER_RR\",\"value\":\"${ICON_AFTER_RR}\"},{\"name\":\"ICON_ANONYMIZE\",\"value\":\"${ICON_ANONYMIZE}\"},{\"name\":\"ICON_VERIFICATION_DATE\",\"value\":\"${ICON_VERIFY_DATE}\"}]}"

          readonly container_overrides="{\"command\": $command_args,\"environment\": $environment_args"
          readonly restore_job_id="$(aws batch submit-job --job-name "$job_name" --job-queue "db-job-queue" \
            --job-definition "db-restore-${ICON_TENANT,,}" \
            --container-overrides "${container_overrides}" | jq -r '.jobId')"

          echo "Job name: ${job_name}"
          echo "Job ID: ${restore_job_id}"

          job_status=''

          echo "Waiting for job: ${job_name}"

          while [[ ! "${job_status}" =~ (SUCCEEDED|FAILED|COMPLETED) ]]; do
              job_status="$(aws batch describe-jobs --jobs "${restore_job_id}" | jq -r '.jobs[0].status')"
              echo "Still waiting for job ${job_name}... Current status is ${job_status}."
              sleep 5
          done
          
          echo "Batch exited with status: ${job_status}"

          if [[ "${jobs_status}" == "FAILED" ]]; then
            echo "Restore failed. Aborting..."
            exit 1
          fi
      - name: Enable batch queue
        if: ${{ inputs.restart }}
        run: |
            echo "Enable batches for tenant: ${ICON_TENANT,,}"
            aws batch update-job-queue --job-queue "icon-queue-${ICON_TENANT,,}" --state "ENABLED" > /dev/null
      - name: Restart iCON services
        if: ${{ inputs.restart }}
        env:
          ICON_ONLINE_REPLICAS: "${{ steps.stop_services.outputs.online_replicas }}"
          ICON_NEMESIS_REPLICAS: "${{ steps.stop_services.outputs.nemesis_replicas }}"
        run: |
            echo "Scaling icon-nemesis-service for tenant ${ICON_TENANT,,} to ${ICON_NEMESIS_REPLICAS}"
            aws ecs update-service --cluster "icon-cluster" --service "icon-nemesis-service-${ICON_TENANT,,}" --desired-count "${ICON_NEMESIS_REPLICAS}" > /dev/null
            aws ecs wait services-stable --cluster "icon-cluster" --service "icon-nemesis-service-${ICON_TENANT,,}"

            echo "Starting icon-online-service for tenant ${ICON_TENANT,,} to ${ICON_ONLINE_REPLICAS}"
            aws ecs update-service --cluster "icon-cluster" --service "icon-online-service-${ICON_TENANT,,}" --desired-count "${ICON_ONLINE_REPLICAS}" > /dev/null
            aws ecs wait services-stable --cluster "icon-cluster" --service "icon-online-service-${ICON_TENANT,,}"
